 56	def known_projects() -> List[Dict[str, str]]:
 57	    """Repositories under DEXTER_CODE_ROOTS (the owner's own checkouts)."""
 58	    out: List[Dict[str, str]] = []
 59	    for root in CODE_ROOTS:
 60	        if os.path.isdir(root) and any(os.path.exists(os.path.join(root, m)) for m in _PROJECT_MARKERS):
 61	            out.append({"name": os.path.basename(root.rstrip("\\/")), "path": root})
 62	        try:
 63	            entries = sorted(os.listdir(root))
 64	        except OSError:
 65	            continue
 66	        for name in entries:
 67	            path = os.path.join(root, name)
 68	            if name.startswith(".") or not os.path.isdir(path):
 69	                continue
 70	            if any(os.path.exists(os.path.join(path, m)) for m in _PROJECT_MARKERS):
 71	                out.append({"name": name, "path": path})
 72	    return out

 75	def resolve_project(hint: Optional[str]) -> Optional[Dict[str, str]]:
 76	    """A known folder name, exact or closest (case-insensitive); paths must be inside a code root."""
 77	    hint = (hint or "").strip().strip("\"'")
 78	    if not hint:
 79	        return None
 80	    projects = known_projects()
 81	    if os.path.isdir(hint):
 82	        full = os.path.abspath(hint)
 83	        for p in projects:
 84	            if os.path.normcase(full) == os.path.normcase(os.path.abspath(p["path"])):
 85	                return p
 86	        return None
 87	    lower = hint.lower().replace(" ", "")
 88	    for p in projects:
 89	        if p["name"].lower() == lower:
 90	            return p
 91	    for p in projects:
 92	        if lower in p["name"].lower() or p["name"].lower() in lower:
 93	            return p
 94	    import difflib
 95	
 96	    names = [p["name"].lower() for p in projects]
 97	    close = difflib.get_close_matches(lower, names, n=1, cutoff=0.6)
 98	    if close:
 99	        return projects[names.index(close[0])]
100	    return None

103	def project_from_text(text: str) -> Optional[Dict[str, str]]:
104	    """The first known project name mentioned anywhere in a sentence."""
105	    lower = (text or "").lower()
106	    best = None
107	    for p in known_projects():
108	        name = p["name"].lower()
109	        if len(name) >= 4 and name in lower:
110	            if best is None or len(name) > len(best["name"]):
111	                best = p
112	    return best

136	    def ingest(self, line: str) -> Optional[Dict[str, Any]]:
137	        """Fold one JSON-lines event into the job; returns a UI-worthy event or None."""
138	        line = line.strip()
139	        if not line:
140	            return None
141	        try:
142	            event = json.loads(line)
143	        except json.JSONDecodeError:
144	            return None
145	        kind = event.get("type")
146	        if kind == "turn_start":
147	            self.turns = int(event.get("turn") or self.turns + 1)
148	        elif kind == "assistant_message":
149	            content = (event.get("content") or "").strip()
150	            if content:
151	                self.final_message = content
152	        elif kind == "tool_call_start":
153	            self.tool_calls += 1
154	            name = str(event.get("name") or "")
155	            args = event.get("arguments") or {}
156	            if name in _EDIT_TOOLS and isinstance(args, dict):
157	                path = args.get("path") or args.get("file") or args.get("file_path")
158	                if path and path not in self.files_changed:
159	                    self.files_changed.append(str(path))
160	            return {"kind": "tool", "name": name, "detail": _short_args(args)}
161	        elif kind == "warning":
162	            self.warnings.append(str(event.get("text") or ""))
163	        elif kind == "error":
164	            self.errors.append(str(event.get("text") or ""))
165	            return {"kind": "error", "detail": str(event.get("text") or "")[:200]}
166	        elif kind == "finished":
167	            self.finish_reason = str(event.get("reason") or "")
168	            self.turns = int(event.get("turns") or self.turns)
169	        elif kind == "verify":
170	            return {"kind": "verify", "ok": bool(event.get("ok")), "detail": str(event.get("command") or "")}
171	        return None

227	    def engine_status(self) -> Dict[str, Any]:
228	        try:
229	            res = requests.get(f"{ENGINE_URL}/models", timeout=2)
230	            if res.ok:
231	                data = res.json().get("data") or []
232	                model = data[0].get("id") if data else None
233	                return {"online": True, "model": model, "url": ENGINE_URL, "startedByDexter": self._engine_proc is not None}
234	        except Exception:
235	            pass
236	        return {"online": False, "model": None, "url": ENGINE_URL, "startedByDexter": self._engine_proc is not None}

238	    def start_engine(self, wait_seconds: int = 90) -> Dict[str, Any]:
239	        status = self.engine_status()
240	        if status["online"]:
241	            return {**status, "started": False}
242	        exe = buzzcode_path()
243	        if not exe:
244	            raise RuntimeError("buzzcode is not installed (expected ~/.cargo/bin/buzzcode.exe).")
245	        os.makedirs(JOB_LOG_DIR, exist_ok=True)
246	        log = open(os.path.join(JOB_LOG_DIR, "engine.log"), "ab")
247	        self._engine_proc = subprocess.Popen(
248	            [exe, "engine", "serve"], stdout=log, stderr=subprocess.STDOUT, stdin=subprocess.DEVNULL, creationflags=_no_window(),
249	        )
250	        deadline = time.time() + wait_seconds
251	        while time.time() < deadline:
252	            if self._engine_proc.poll() is not None:
253	                raise RuntimeError("buzzcode engine exited while starting; see data/code_jobs/engine.log.")
254	            status = self.engine_status()
255	            if status["online"]:
256	                return {**status, "started": True}
257	            time.sleep(2)
258	        return {**self.engine_status(), "started": True, "warming": True}
259	
260	    def stop_engine(self) -> bool:
261	        if self._engine_proc is None:
262	            return False
263	        try:
264	            self._engine_proc.terminate()
265	        except Exception:
266	            pass
267	        self._engine_proc = None
268	        return True

275	    def start(self, task: str, project: Dict[str, str], mode: str = "edits") -> CodeJob:
276	        task = (task or "").strip()
277	        if not task:
278	            raise ValueError("What should be coded?")
279	        mode = MODES.get((mode or "edits").lower())
280	        if mode is None:
281	            raise ValueError("Mode must be 'edits' (files only) or 'yolo' (files and commands).")
282	        exe = buzzcode_path()
283	        if not exe:
284	            raise RuntimeError("buzzcode is not installed (expected ~/.cargo/bin/buzzcode.exe).")
285	        if self.running():
286	            current = self.running()[0]
287	            raise RuntimeError(f"A code task is already running ({current.id}: {current.task[:60]}). Wait for it or cancel it.")
288	        job = CodeJob(task, project, mode)
289	        os.makedirs(JOB_LOG_DIR, exist_ok=True)
290	        env = dict(os.environ)
291	        env.setdefault("NO_COLOR", "1")
292	        job.process = subprocess.Popen(
293	            [exe, "-C", project["path"], "--mode", mode, "--json", "-p", task],
294	            stdout=subprocess.PIPE, stderr=subprocess.PIPE, stdin=subprocess.DEVNULL,
295	            creationflags=_no_window(), env=env, cwd=project["path"],
296	        )
297	        with self._lock:
298	            self.jobs[job.id] = job
299	        threading.Thread(target=self._follow, args=(job,), name=f"dexter-code-{job.id}", daemon=True).start()
300	        bus.publish("code", {"jobId": job.id, "status": "started", "task": task, "project": project["name"]})
301	        return job

303	    def _follow(self, job: CodeJob) -> None:
304	        proc = job.process
305	        assert proc is not None and proc.stdout is not None
306	        last_publish = 0.0
307	        with open(job.log_path, "a", encoding="utf-8") as log:
308	            for raw in iter(proc.stdout.readline, b""):
309	                line = raw.decode("utf-8", errors="replace")
310	                log.write(line)
311	                notable = job.ingest(line)
312	                now = time.time()
313	                if notable and now - last_publish > 1.5:
314	                    last_publish = now
315	                    bus.publish("code", {"jobId": job.id, "status": "running", **notable, "toolCalls": job.tool_calls})
316	        stderr = b""
317	        try:
318	            stderr = proc.stderr.read() if proc.stderr else b""
319	        except Exception:
320	            pass
321	        job.exit_code = proc.wait()
322	        job.finished = time.time()
323	        if stderr.strip():
324	            job.errors.append(stderr.decode("utf-8", errors="replace").strip()[-800:])
325	        if job.status == "running":
326	            job.status = "done" if job.exit_code == 0 and (job.final_message or not job.errors) else "failed"
327	        job.diff_stat = self._diff_stat(job.project["path"])
328	        self._remember(job)
329	        self._announce(job)

331	    def _diff_stat(self, path: str) -> str:
332	        if not os.path.isdir(os.path.join(path, ".git")):
333	            return ""
334	        try:
335	            out = subprocess.run(
336	                ["git", "diff", "--stat"], cwd=path, capture_output=True, text=True, timeout=15, creationflags=_no_window(),

363	    def cancel(self, job_id: Optional[str] = None) -> Optional[CodeJob]:
364	        job = self.jobs.get(job_id) if job_id else (self.running()[0] if self.running() else None)
365	        if not job or job.status != "running":
366	            return None
367	        job.status = "cancelled"
368	        try:

375	    def status(self, job_id: Optional[str] = None) -> Optional[Dict[str, Any]]:
376	        if job_id and job_id in self.jobs:
377	            return self.jobs[job_id].to_dict()
378	        if self.running():

380	        latest = max(self.jobs.values(), key=lambda j: j.started, default=None)

385	    def recent(self, limit: int = 10) -> List[Dict[str, Any]]:
386	        live = sorted((j.to_dict() for j in self.jobs.values()), key=lambda d: d["started"], reverse=True)
