  9	type SubTab = 'discovered' | 'saved';

 21

 23	
 24	  const loadSaved = async () => {

 26	      const data = await getJson<{ topics?: Topic[] }>('/api/topics/saved');
 27	      setSaved(data.topics || []);

 29	      toast(describeError(err), 'error');
 30	    }
 31	  };
 32

 36	      const data = await postJson<{ topics?: Topic[]; source?: string }>('/api/topics/discover', { channel });
 37	      setDiscovered(data.topics || []);

 39	    } catch (err) {
 40	      setDiscovered([]);
 41	      toast(describeError(err), 'error');
 42	    } finally {
 43	      setDiscovering(false);
 44	    }
 45	  };
 46	
 47	  useEffect(() => {
 48	    discover(selectedChannel);
 49	    loadSaved();
 50	    // eslint-disable-next-line react-hooks/exhaustive-deps
 51	  }, []);
 52	
 53	  const pickChannel = (id: string) => {
 54	    setSelectedChannel(id);
 55	    discover(id);
 56	  };
 57	
 58	  const save = async (item: Topic) => {
 59	    try {
 60	      await postJson('/api/topics/save', {
 61	        topic: item.topic,
 62	        category: item.category || 'General',
 63	        channel: item.channel || selectedChannel,
 64	        viral_potential: item.viral_potential || 0,
 65	        country: item.country || '',
 66	        source_type: item.source_type || '',
 67	        sources_used: item.sources_used || [],
 68	        visual_requirements: item.visual_requirements || [],
 69	        exclusion_audit: item.exclusion_audit || '',
 70	        notes: item.notes || '',
 71	      });
 72	      toast(`Saved "${item.topic.slice(0, 40)}" to the vault.`, 'success');
 73	      loadSaved();
 74	    } catch (err) {
 75	      toast(describeError(err), 'error');
 76	    }
 77	  };
 78	
 79	  const remove = async (item: Topic) => {
 80	    const yes = await confirm({ title: 'Delete this saved topic?', body: item.topic, confirmLabel: 'Delete', danger: true });
 81	    if (!yes) return;
 82	    try {
 83	      await postJson('/api/topics/delete', { id: item.id, topic: item.topic });
 84	      loadSaved();
 85	    } catch (err) {
 86	      toast(describeError(err), 'error');
 87	    }
 88	  };
 89	
 90	  const discuss = (item: Topic) => {
 91	    try {
 92	      sessionStorage.setItem(CHAT_TOPIC_KEY, JSON.stringify({ ...item, channel: item.channel || selectedChannel }));
 93	    } catch {
 94	      // Session storage unavailable; the chat still opens.
 95	    }
 96	    navigate('studio_chat');
 97	  };
 98	
 99	  const develop = async (item: Topic) => {
100	    const brand = item.channel || selectedChannel;
101	    try {
102	      await postJson('/api/projects', { name: item.topic, brand, workflow_name: workflowFor(brand) });
103	      toast(`Created a ${brand} project for "${item.topic.slice(0, 40)}".`, 'success');
104	      await refreshProjects();
105	      navigate('projects');
106	    } catch (err) {
107	      toast(`Could not create the project: ${describeError(err)}`, 'error');
108	    }
109	  };
110	
111	  const visibleSaved = filter === 'all' ? saved : saved.filter((t) => (t.channel || '').toLowerCase().includes(filter.toLowerCase()));
112	
113	  const card = (item: Topic, kind: SubTab, idx: number) => (
114	    <div key={item.id || `${kind}-${idx}`} className="panel-card" style={{ padding: 24, margin: 0, display: 'flex', flexDirection: 'column', gap: 16, borderLeft: `4px solid ${kind === 'saved' ? '#f59e0b' : '#f43f5e'}` }}>
115	      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start', flexWrap: 'wrap', gap: 12 }}>
116	        <div>
117	          <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 6, flexWrap: 'wrap' }}>
118	            {item.channel && <span className="brand-badge">{item.channel}</span>}
119	            {item.category && <span className="brand-badge" style={{ backgroundColor: '#1f1a29', color: '#a78bfa' }}>{item.category}</span>}
120	            {typeof item.viral_potential === 'number' && (
121	              <span style={{ fontSize: '0.8rem', color: '#4ade80', backgroundColor: '#0b261d', padding: '2px 8px', borderRadius: 4, fontWeight: 600 }}>Viral score {item.viral_potential}/10</span>
122	            )}
123	            {item.country && <span style={{ fontSize: '0.8rem', color: '#38bdf8', backgroundColor: '#0c2233', padding: '2px 8px', borderRadius: 4 }}>{item.country}</span>}
124	          </div>
125	          <h4 style={{ margin: 0, fontSize: '1.2rem', color: '#ffffff' }}>{item.topic}</h4>
126	        </div>
127	        <div style={{ display: 'flex', gap: 10, flexWrap: 'wrap' }}>
128	          {kind === 'discovered' && (
129	            <button type="button" className="btn btn-outline" style={{ padding: '8px 14px', fontSize: '0.85rem' }} onClick={() => save(item)}>
130	              <Bookmark size={15} style={{ color: '#f59e0b' }} />
131	              <span>Save</span>
132	            </button>
133	          )}
134	          <button type="button" className="btn btn-outline" style={{ padding: '8px 14px', fontSize: '0.85rem' }} onClick={() => discuss(item)}>
135	            <MessageSquare size={15} style={{ color: '#a78bfa' }} />
136	            <span>Discuss</span>
137	          </button>
138	          <button type="button" className="btn" style={{ padding: '8px 16px', fontSize: '0.85rem' }} onClick={() => develop(item)}>
139	            <Send size={15} />
140	            <span>Start project</span>
141	          </button>
142	          {kind === 'saved' && (
143	            <button type="button" className="btn btn-outline" style={{ padding: '8px 12px', fontSize: '0.85rem', borderColor: '#7f1d1d', color: '#fca5a5' }} onClick={() => remove(item)} aria-label="Delete">
144	              <Trash2 size={15} />
145	            </button>
146	          )}
147	        </div>
148	      </div>
149	      <div style={{ fontSize: '0.9rem', color: '#cbd5e1', display: 'flex', flexDirection: 'column', gap: 6 }}>
150	        {item.source_type && <div><strong>Source type:</strong> {item.source_type}</div>}
151	        {list(item.sources_used) && <div><strong>Sources:</strong> {list(item.sources_used)}</div>}
152	        {list(item.visual_requirements) && <div><strong>Visuals:</strong> {list(item.visual_requirements)}</div>}
153	        {item.exclusion_audit && <div style={{ fontSize: '0.85rem', color: '#4ade80', fontWeight: 600 }}>{item.exclusion_audit}</div>}
154	        {item.notes && <div style={{ fontStyle: 'italic', color: '#94a3b8' }}>{item.notes}</div>}
155	      </div>
156	    </div>

159	  return (
160	    <div>

162	        <button type="button" className={`subtab-btn ${sub === 'discovered' ? 'active' : ''}`} onClick={() => setSub('discovered')}>
163	          <Compass size={18} />
164	          <span>Topic ideas</span>
165	        </button>
166	        <button type="button" className={`subtab-btn ${sub === 'saved' ? 'active' : ''}`} onClick={() => { setSub('saved'); loadSaved(); }}>
167	          <Star size={18} style={{ color: '#f59e0b' }} />
168	          <span>Saved vault ({saved.length})</span>

175	            <h3 className="panel-title"><Compass size={22} /> Topic ideas by channel</h3>

177	              {channels.map((ch) => (
178	                <div
179	                  key={ch.id}
180	                  onClick={() => pickChannel(ch.id)}
181	                  style={{ flex: 1, minWidth: 200, padding: 16, borderRadius: 10, border: selectedChannel === ch.id ? '2px solid #f43f5e' : '1px solid #1e2230', backgroundColor: selectedChannel === ch.id ? '#1c1724' : '#12141d', cursor: 'pointer' }}
182	                >
183	                  <div style={{ fontSize: '1.1rem', marginBottom: 4 }}>{ch.icon} <strong style={{ color: '#ffffff' }}>{ch.id}</strong></div>

190	                <Sparkles size={18} />

195	                  Written by {metaFor(selectedChannel).strategist}
196	                </span>

211	          </div>

218	            <div>

221	            </div>

223	              <option value="all">All channels ({saved.length})</option>
224	              {channels.map((ch) => <option key={ch.id} value={ch.id}>{ch.id}</option>)}
225	            </select>
226	          </div>
227	          <div style={{ display: 'flex', flexDirection: 'column', gap: 20 }}>
228	            {visibleSaved.length === 0 ? (
229	              <div className="panel-card" style={{ textAlign: 'center', padding: 40, color: '#94a3b8' }}>
230	                <Bookmark size={36} style={{ marginBottom: 12, opacity: 0.5 }} />
231	                <p style={{ margin: 0 }}>Nothing saved for this filter.</p>
232	              </div>
233	            ) : (
234	              visibleSaved.map((item, idx) => card(item, 'saved', idx))
235	            )}
236	          </div>
237	        </div>
