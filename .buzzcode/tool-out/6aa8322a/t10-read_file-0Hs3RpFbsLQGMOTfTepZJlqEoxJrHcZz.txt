  1	"""
  2	Check that the Buzzcaf Studio can actually run before opening its window.
  3	
  4	    python backend/preflight.py           full report, exit 1 on a fatal problem
  5	    python backend/preflight.py --quiet   only print problems
  6	
  7	desktop_app.py calls run_checks() itself and shows the failures in a dialog,
  8	because under pythonw there is no console for a report to land in.
  9	
 10	FAIL rows stop the Studio from working at all; WARN rows describe a degraded
 11	capability (no LLM provider reachable, a missing persona) and never block.
 12	"""

 23	OK, WARN, FAIL = "OK", "WARN", "FAIL"
 24	_SYMBOL = {OK: "[ OK ]", WARN: "[WARN]", FAIL: "[FAIL]"}

 27	    ("fastapi", "the API server"),
 28	    ("uvicorn", "the API server"),
 29	    ("pydantic", "request schemas"),

 37	class Report:
 38	    def __init__(self, quiet: bool = False) -> None:
 39	        self.rows = []
 40	        self.quiet = quiet
 41	
 42	    def add(self, status: str, title: str, detail: str = "", action: str = "") -> None:
 43	        self.rows.append((status, title, detail, action))
 44	        if self.quiet and status == OK:
 45	            return
 46	        line = f"{_SYMBOL[status]} {title}"
 47	        if detail:
 48	            line += f" - {detail}"
 49	        print(line)
 50	        if action and status != OK:
 51	            print(f"       -> {action}")
 52	
 53	    @property
 54	    def failed(self) -> bool:
 55	        return any(row[0] == FAIL for row in self.rows)
 56	
 57	    def problem_lines(self):
 58	        lines = []
 59	        for status, title, detail, action in self.rows:
 60	            if status == OK:
 61	                continue
 62	            line = f"{_SYMBOL[status]} {title}"
 63	            if detail:
 64	                line += f" - {detail}"
 65	            if action:
 66	                line += f"  -> {action}"
 67	            lines.append(line)
 68	        return lines
 69	
 70	    def summary(self) -> str:
 71	        counts = {OK: 0, WARN: 0, FAIL: 0}
 72	        for status, *_ in self.rows:
 73	            counts[status] += 1
 74	        return f"{counts[OK]} ok, {counts[WARN]} warning(s), {counts[FAIL]} failure(s)"

 77	def _installed(module: str) -> bool:
 78	    try:
 79	        return importlib.util.find_spec(module) is not None
 80	    except Exception:
 81	        return False

 84	def check_python_deps(report: Report) -> None:
 85	    missing = [(m, why) for m, why in _REQUIRED if not _installed(m)]
 86	    if missing:
 87	        report.add(FAIL, "Python dependencies", "missing " + ", ".join(m for m, _ in missing),
 88	                   "pip install -r requirements.txt")
 89	    else:
 90	        report.add(OK, "Python dependencies", "core packages present")

113	def check_paths(report: Report) -> None:
114	    try:
115	        from core import paths
116	
117	        problems = [
118	            name for name, value in (
119	                ("prompts", paths.PROMPTS_DIR), ("knowledge", paths.KNOWLEDGE_DIR),
120	                ("projects", paths.PROJECTS_DIR),
121	            ) if not os.path.isdir(value)
122	        ]

124	            report.add(WARN, "Data directories", "missing: " + ", ".join(problems),
125	                       "They are created on first use; check PROJECTS_PATH/KNOWLEDGE_PATH in .env if unexpected")
126	        else:

132	def check_personas(report: Report) -> None:
133	    try:
134	        from core.paths import AGENTS_DIR
135	
136	        count = len([f for f in os.listdir(AGENTS_DIR) if f.endswith(".md")])
137	    except Exception as exc:
138	        report.add(WARN, "Agent personas", f"could not count: {exc}")
139	        return
140	    if count == 0:
141	        report.add(FAIL, "Agent personas", "no persona files found in prompts/agents")
142	    elif count != EXPECTED_PERSONAS:
143	        report.add(WARN, "Agent personas", f"{count} found, expected {EXPECTED_PERSONAS}")
144	    else:
145	        report.add(OK, "Agent personas", f"{count} personas registered")

148	def check_port(report: Report) -> None:
149	    """Report on the preferred port. Never fatal: if an unrelated process holds
150	    it, the Studio steps to the next free port at launch (desktop_app._resolve_port)."""
151	    import json
152	    import socket
153	    import urllib.request
154	
155	    port = int(os.getenv("BUZZCAF_PORT", "8099"))
156	    try:
157	        req = urllib.request.Request(f"http://127.0.0.1:{port}/health")
158	        with urllib.request.urlopen(req, timeout=1.5) as res:
159	            body = json.loads(res.read().decode("utf-8") or "{}")
160	        if body.get("app") == "buzzcaf":
161	            report.add(OK, "Port", f"{port} already served by a running Studio (window will attach)")
162	        else:
163	            report.add(WARN, "Port", f"{port} is taken by something that is not the Studio",
164	                       "the Studio will start on the next free port")
165	        return
166	    except Exception:
167	        pass
168	    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
169	        try:
170	            sock.bind(("127.0.0.1", port))
171	            report.add(OK, "Port", f"{port} is free")
172	        except OSError:
173	            report.add(WARN, "Port", f"{port} is in use but does not answer /health",
174	                       "the Studio will start on the next free port")

177	def check_llm(report: Report) -> None:
178	    """At least one provider should be reachable; the Studio still boots without."""
179	    try:
180	        import requests
181	        from integrations.llm import load_config
182	
183	        cfg = load_config()
184	    except Exception as exc:
185	        report.add(WARN, "LLM providers", f"could not read config: {exc}")
186	        return
187	
188	    reachable = []
189	    lm_url = (cfg.get("lm_studio_url") or "http://localhost:1234/v1").rstrip("/")
190	    try:
191	        if requests.get(f"{lm_url}/models", timeout=1.5).status_code == 200:
192	            reachable.append("LM Studio")
193	    except Exception:
194	        pass
195	    try:
196	        if requests.get("http://127.0.0.1:11434/v1/models", timeout=1.5).status_code == 200:
197	            reachable.append("Ollama")
198	    except Exception:
199	        pass
200	    if cfg.get("gemini_api_key"):
201	        reachable.append("Gemini (key set)")
202	    if cfg.get("openai_api_key"):
203	        reachable.append("OpenAI (key set)")
204	    if reachable:
205	        report.add(OK, "LLM providers", ", ".join(reachable))
206	    else:
207	        report.add(WARN, "LLM providers", "none reachable; replies will be labelled SIMULATED",
208	                   "Start LM Studio or Ollama, or add a key in Settings")

211	def run_fast_checks(quiet: bool = True, dev: bool = False) -> Report:
212	    """The checks that can FAIL and answer in milliseconds; they gate the window."""
213	    report = Report(quiet=quiet)
214	    check_python_deps(report)
215	    check_desktop_shell(report)
216	    check_ui_bundle(report, dev)
217	    check_paths(report)
218	    check_personas(report)
219	    check_port(report)
220	    return report

223	def run_slow_checks(quiet: bool = True) -> Report:
224	    """Network probes that only ever WARN; the desktop shell runs them after the window opens."""
225	    report = Report(quiet=quiet)
226	    check_llm(report)
227	    return report
228	
229	
230	def run_checks(quiet: bool = True, dev: bool = False) -> Report:
231	    report = run_fast_checks(quiet=quiet, dev=dev)
232	    for row in run_slow_checks(quiet=True).rows:
233	        report.add(*row)
234	    return report

237	def main() -> int:
238	    parser = argparse.ArgumentParser(description="Check that the Buzzcaf Studio can run.")
239	    parser.add_argument("--quiet", action="store_true", help="only print problems")
240	    parser.add_argument("--dev", action="store_true", help="developer mode (Vite dev server)")
241	    args = parser.parse_args()
242	    if not args.quiet:
243	        print("=== Buzzcaf Studio preflight ===")
244	    report = run_checks(quiet=args.quiet, dev=args.dev)
245	    print(f"--- {report.summary()} ---")
246	    if report.failed:
247	        print("The Studio will not start correctly until the failures above are fixed.")
248	        return 1
249	    return 0
