 76	def _setup_logging() -> str:
 77	    try:
 78	        from core.paths import LOGS_DIR
 79	
 80	        log_dir = LOGS_DIR
 81	    except Exception:
 82	        log_dir = os.path.join(BACKEND_DIR, "logs")
 83	    os.makedirs(log_dir, exist_ok=True)
 84	    log_path = os.path.join(log_dir, "studio.log")
 85	
 86	    from logging.handlers import RotatingFileHandler
 87	
 88	    handler = RotatingFileHandler(log_path, maxBytes=2_000_000, backupCount=3, encoding="utf-8")
 89	    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(name)s: %(message)s"))
 90	    root = logging.getLogger()
 91	    root.setLevel(logging.INFO)
 92	    root.addHandler(handler)
 93	
 94	    # pythonw starts with stdout/stderr set to None; anything that writes to
 95	    # them (uvicorn's access log, a traceback) would be lost or raise.
 96	    if sys.stdout is None or sys.stderr is None:
 97	        stream = open(log_path, "a", encoding="utf-8", buffering=1)
 98	        if sys.stdout is None:
 99	            sys.stdout = stream
100	        if sys.stderr is None:
101	            sys.stderr = stream
102	    return log_path

123	def _studio_answering(url: str = HEALTH_URL, timeout: float = 1.5) -> bool:
124	    """True only if *this app* answers - anything else on the port is not ours."""
125	    try:
126	        import json
127	
128	        req = urllib.request.Request(url, headers={"User-Agent": "BuzzcafStudio/desktop"})
129	        with urllib.request.urlopen(req, timeout=timeout) as res:
130	            if res.status != 200:
131	                return False
132	            body = json.loads(res.read().decode("utf-8") or "{}")
133	            return body.get("app") == "buzzcaf"
134	    except Exception:
135	        return False

150	def _write_runtime(port: int) -> None:
151	    """Tell the neighbours where the Studio actually landed.
152	
153	    Dexter reads this file (through BUZZCAF_DIR) before probing, so a Studio
154	    that had to step off its usual port is still found instead of reported
155	    offline - and nothing else on 8099 gets mistaken for us, because readers
156	    still check that /health says app:"buzzcaf".
157	    """
158	    import json
159	
160	    try:
161	        os.makedirs(os.path.dirname(RUNTIME_FILE), exist_ok=True)
162	        tmp = RUNTIME_FILE + ".tmp"
163	        with open(tmp, "w", encoding="utf-8") as handle:
164	            json.dump({
165	                "port": port, "pid": os.getpid(), "started_at": time.time(),
166	                "health_url": f"http://{HOST}:{port}/health",
167	            }, handle)
168	        os.replace(tmp, RUNTIME_FILE)
169	    except Exception as exc:
170	        log.warning("Could not write %s: %s", RUNTIME_FILE, exc)

201	def _resolve_port(preferred: int) -> Tuple[int, bool]:
202	    """Decide which port the backend will use.
203	
204	    Returns (port, ours_already_answering):
205	      - If a Studio is already up anywhere discovery can reach it (an explicit
206	        BUZZCAF_URL, the shared ledger, or a scan of preferred…preferred+span
207	        that checks /health says app:"buzzcaf"), attach to it - a window just
208	        opens on it rather than a second backend starting.
209	      - Else take the preferred port, or step forward past whatever unrelated
210	        process holds it. Never evict: `pick_port` only ever binds what is free.
211	    """
212	    from urllib.parse import urlparse
213	
214	    # ttl=0: a launcher must not act on a three-second-old answer.
215	    running = buzzcaf_ports.discover(APP_ID, preferred, span=PORT_SCAN_RANGE, host=HOST, ttl=0)
216	    if running:
217	        port = urlparse(running).port or preferred
218	        if port != preferred:
219	            log.info("A running Studio answers on %d (discovered); attaching.", port)
220	        return port, True
221	    # The runtime file can point outside the scan window (an OS-assigned port
222	    # from a launch when everything was busy), so it is still worth a look.
223	    recorded = _read_runtime().get("port")
224	    if isinstance(recorded, int) and _studio_answering(f"http://{HOST}:{recorded}/health"):
225	        log.info("A running Studio answers on %d (from %s); attaching.", recorded, RUNTIME_FILE)
226	        return recorded, True
227	
228	    port = buzzcaf_ports.pick_port(preferred, PORT_SCAN_RANGE, HOST)
229	    if port != preferred:
230	        log.warning("Port %d is in use by another process; using %d instead.", preferred, port)
231	    return port, False

323	def _start_vite_dev(port: int, dev_port: int) -> Optional[subprocess.Popen]:
324	    """`npm run dev`, hidden, for --dev. Returns the process so it dies with us.
325	
326	    The real backend port travels in the child's environment: vite.config.ts
327	    points its proxy at BUZZCAF_PORT, and app.main re-reads .env with override
328	    on, so os.environ alone cannot be trusted by the time Vite starts. The dev
329	    port is pinned with --strictPort so Vite fails loudly instead of moving.
330	    """
331	    npm = shutil.which("npm")
332	    if not npm:
333	        _message_box("npm not found", "Developer mode needs Node.js (npm) on PATH to run the Vite dev server.")
334	        return None
335	    try:
336	        return subprocess.Popen(
337	            [npm, "run", "dev", "--", "--port", str(dev_port), "--strictPort"], cwd=FRONTEND_DIR,
338	            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, creationflags=_hidden_flags(),
339	            env={**os.environ, "BUZZCAF_PORT": str(port), "BUZZCAF_DEV_PORT": str(dev_port)},
340	        )
341	    except Exception as exc:
342	        _message_box("Could not start the Vite dev server", str(exc))

346	def _stop_process_tree(proc: subprocess.Popen, port: Optional[int] = None) -> None:
347	    """End a child and everything it spawned, and free the port it was given.
348	
349	    `npm run dev` is cmd.exe (the npm shim) over node over another shell over
350	    node (Vite); terminate() kills only the shim, and even taskkill /T misses
351	    a level once a shim has exited, leaving Vite holding its port for the
352	    next launch to trip over. So: the whole tree via psutil when available,
353	    then whatever still listens on our port - we chose a free one, so any
354	    listener there is ours - then taskkill as the last resort.
355	    """
356	    try:
357	        import psutil
358	
359	        procs = []
360	        try:
361	            root = psutil.Process(proc.pid)
362	            procs = root.children(recursive=True) + [root]
363	        except psutil.Error:
364	            pass
365	        if port is not None:
366	            try:
367	                for conn in psutil.net_connections(kind="tcp"):
368	                    if conn.status == psutil.CONN_LISTEN and conn.laddr.port == port and conn.pid:
369	                        procs.append(psutil.Process(conn.pid))
370	            except psutil.Error:
371	                pass
372	        for item in procs:
373	            try:
374	                item.kill()
375	            except psutil.Error:
376	                pass
377	        if procs:
378	            psutil.wait_procs(procs, timeout=3)
379	            return
380	    except ImportError:
381	        pass
382	    except Exception as exc:
383	        log.warning("Process tree cleanup fell back to taskkill: %s", exc)
384	    try:
385	        if sys.platform == "win32":
386	            subprocess.run(["taskkill", "/T", "/F", "/PID", str(proc.pid)],
387	                           stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, creationflags=_hidden_flags())
388	        else:
389	            proc.terminate()
390	    except Exception:
391	        pass
