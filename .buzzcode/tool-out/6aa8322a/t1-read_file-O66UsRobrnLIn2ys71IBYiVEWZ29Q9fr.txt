348	347	    """End a child and everything it spawned, and free the port it was given.
349	348	
350	349	    `npm run dev` is cmd.exe (the npm shim) over node over another shell over
351	350	    node (Vite); terminate() kills only the shim, and even taskkill /T misses
352	351	    a level once a shim has exited, leaving Vite holding its port for the
353	352	    next launch to trip over. So: the whole tree via psutil when available,
354	353	    then whatever still listens on our port - we chose a free one, so any
355	354	    listener there is ours - then taskkill as the last resort.
356	355	    """
357	356	    try:
358	357	        import psutil
359	358	
360	359	        procs = []

363	362	            procs = root.children(recursive=True) + [root]
364	363	        except psutil.Error:

369	368	                    if conn.status == psutil.CONN_LISTEN and conn.laddr.port == port and conn.pid:
